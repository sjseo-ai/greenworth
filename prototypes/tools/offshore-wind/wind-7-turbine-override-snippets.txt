  // 빈 칸·0 이하·숫자가 아닌 값은 "자동"으로 본다(칸을 비우면 언제든 모델 계산값으로 돌아온다).
  function turbineOverride(id, fld) {
    const el = $(`tb-${id}-${fld}`);
    if (!el) return null;
    const raw = String(el.value).trim();
    if (raw === '') return null;
    const v = Number(raw);
    return Number.isFinite(v) && v > 0 ? v : null;
  }
  function turbineDerived(id) {
    const sp = turbineSpec(id);
    const specificPower = sp.rotor > 0 ? sp.mw * 1e6 / (Math.PI * (sp.rotor / 2) ** 2) : 0;
    const curveF = specificPower > 0 ? ((+$('refSpecificPower').value || specificPower) / specificPower) ** (+$('powerCurveExp').value || 0) : 1;
    const losses = (1 - rate(+$('wakeLossPct').value || 0)) * (1 - rate(+$('electricalLossPct').value || 0));
    // 보증 주체별 곱 — 터빈사(출력곡선 보증 이행률 × 기술적 가용률) × EPC·O&M(BOP·계통 가용률)
    const warranty = rate(+$('powerCurveWarrantyPct').value || 0) * rate(sp.avail) * rate(+$('bopAvailPct').value || 0);
    const y1Auto = Math.round((+$('siteBaseRatePct').value || 0) * curveF * warranty * losses * 100) / 100;
    const epcAuto = Math.round((sp.price + (+$('bosUnitPrice').value || 0)) * 10) / 10;
    const y1Ov = turbineOverride(id, 'y1ov'), epcOv = turbineOverride(id, 'epcov');
    const y1 = y1Ov ?? y1Auto, epc = epcOv ?? epcAuto;
    // y1Scale — 터빈사 제공값이 모델 계산값의 몇 배인지. 보증·LD 탭이 단계별 분해를 이 배율로 맞춰,
    // 보증 기준 이용률이 화면의 1년차 순이용률과 어긋나지 않게 한다.
    return { ...sp, specificPower, curveF, y1, y1Auto, y1Ov: y1Ov != null, epc, epcAuto, epcOv: epcOv != null,
      y1Scale: y1Auto > 0 ? y1 / y1Auto : 1 };
  }
