  // 자금조달 — 비율 모드: 총사업비 × 자기자본·주민채권 비율 / 금액 모드: 자기자본·주민채권을 억원으로 고정 /
  // DSCR 모드: 선순위를 목표 DSCR로 먼저 정하고(sizedSeniorAmount) 자기자본이 나머지를 메운다.
  // 비율·금액 모드에서는 선순위 대출 = 총사업비 − 자기자본 − 주민채권(총사업비가 금융비용 때문에 반복 수렴하는 동안 선순위가 흡수한다).
  function fundingSplit(model, total) {
    const f = model.finance;
    if (f.fundingMode === 'dscr') {
      const bond = total * rate(f.residentBondPct);
      const senior = Math.max(0, Math.min(Math.max(0, total - bond), f.sizedSeniorAmount ?? 0));
      return { equity: Math.max(0, total - bond - senior), bond };
    }
    if (f.fundingMode === 'amount') return { equity: Math.max(0, f.equityAmount || 0), bond: Math.max(0, f.bondAmount || 0) };
    return { equity: total * rate(f.equityPct), bond: total * rate(f.residentBondPct) };
  }

  // ---------- 목표 DSCR 기반 선순위 부채 사이징 (project finance debt sizing) ----------
  // 실무 순서: 보수적 발전량(보통 P90)의 CFADS를 목표 DSCR로 나눈 금액까지만 연간 원리금으로 쓰고,
  // 그 원리금을 감당할 수 있는 원금이 선순위 한도가 된다. 자기자본은 총사업비에서 선순위·주민채권을 뺀 나머지.
  // CFADS와 부채는 서로 영향을 주므로(이자 절세 · 금융수수료 · 건설이자 · DSRA가 총사업비를 움직인다) 몇 번 반복해 수렴시킨다.
  // 원금 1억당 그 해 선순위 원리금 = 거치기간에는 금리, 상환기간에는 원리금균등 계수 — 부분연도는 가동일수로 안분한다.
  // DSCR 원리금 범위 — 기본은 "선순위 원리금만"(실무의 senior DSCR). 주민참여채권까지 넣으면 만기 일시상환이 있는
  // 한 해가 DSCR을 짓눌러 그 해 하나가 전체 부채 규모를 지배하므로, 그 기준이 필요할 때만 고르도록 선택지로 두었다.
  function seniorCapFromDscr(model, detail, target) {
    const f = model.finance;
    const scopeAll = f.dscrDebtScope === 'all';
    const r1 = seniorRateForYear(model, 1);
    const grace = Math.max(0, Math.round(f.seniorGraceYears) || 0);
    const term = Math.max(1, Math.round(f.seniorTermYears) || 1);
    const n = Math.max(1, term - grace);
    const perUnitRepay = r1 > 0 ? r1 / (1 - (1 + r1) ** -n) : 1 / n;
    const opRows = detail.filter((d) => d.operatingDays > 0);
    let cap = Infinity, minSenior = Infinity, minAll = Infinity;
    opRows.forEach((d, i) => {
      const seq = i + 1;
      if (seq > term) return;
      const frac = d.daysInYear > 0 ? d.operatingDays / d.daysInYear : 1;
      const room = d.cfads / target - (scopeAll ? d.bondInterest + d.bondPrincipal : 0); // 선순위 원리금으로 쓸 수 있는 금액
      const perUnit = (seq <= grace ? r1 : perUnitRepay) * frac;
      if (perUnit <= 1e-12) return;
      cap = Math.min(cap, room / perUnit);
      const seniorDs = d.seniorInterest + d.seniorPrincipal;
      if (seniorDs > 1e-9) minSenior = Math.min(minSenior, d.cfads / seniorDs);
      if (d.dscr != null) minAll = Math.min(minAll, d.dscr);
    });
    const fin = (v) => (Number.isFinite(v) ? v : NaN);
    return { cap: Number.isFinite(cap) ? Math.max(0, cap) : 0,
      minDscr: scopeAll ? fin(minAll) : fin(minSenior), minSenior: fin(minSenior), minAll: fin(minAll) };
  }

  function withSizedSenior(model) {
    const f = model.finance;
    const target = Math.max(0.01, +f.targetDscrSizing || 1.3);
    // 사이징 기준 발전량 — 확률수준을 고르면 그 수준의 연차별 순이용률로 CFADS를 본다(기본은 현재 적용값).
    const probeBase = cloneModel(model);
    probeBase.finance.sizedSeniorReady = true;
    if (Array.isArray(f.dscrSizingRates) && f.dscrSizingRates.length) probeBase.revenue.operatingRatesPct = f.dscrSizingRates;
    // 출발점 — 비율 모드로 구한 선순위(수렴을 빠르게 하기 위한 씨앗값)
    const seed = cloneModel(probeBase);
    seed.finance.fundingMode = 'ratio';
    let senior = Math.max(0, solveInvestment(seed, buildCalendar(seed.project)).seniorPrincipal);
    let ok = false;
    for (let i = 0; i < 6; i++) {
      const probe = cloneModel(probeBase);
      probe.finance.sizedSeniorAmount = senior;
      const r = analyzeEss(probe);
      if (r.errors && r.errors.length) break;
      ok = true;
      const next = seniorCapFromDscr(probe, r.detail, target).cap;
      const done = Math.abs(next - senior) < 0.05;
      senior = next;
      if (done) break;
    }
    // 확정된 한도로 한 번 더 돌려 실제 최소 DSCR을 확인한다(사이징 기준 DSCR은 목표와 같아야 정상).
    let minDscr = NaN, minSenior = NaN, minAll = NaN;
    if (ok) {
      const verify = cloneModel(probeBase);
      verify.finance.sizedSeniorAmount = senior;
      const rv = analyzeEss(verify);
      if (!(rv.errors && rv.errors.length)) ({ minDscr, minSenior, minAll } = seniorCapFromDscr(verify, rv.detail, target));
    }
    lastDscrSizing = { senior, target, minDscr, minSenior, minAll, scopeAll: f.dscrDebtScope === 'all', ok };
    const sized = cloneModel(model);
    sized.finance.sizedSeniorAmount = senior;
    sized.finance.sizedSeniorReady = true;
    return sized;
  }
