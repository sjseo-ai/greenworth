    // ── 준공지연 LD ──
    const weeks = Math.max(0, +$('delayWeeks').value || 0);
    const epcContract = (+$('epcUnitPrice').value || 0) * mw; // ① 공사비(도급액) 억원 = EPC 단가(억원/MW) × MW
    const annualRevenue = moneyOf(genOf(cfGuar));
    const delayLoss = Number.isFinite(annualRevenue) ? (annualRevenue / 365) * weeks * 7 : NaN;
    const ldDelayRaw = epcContract * (num('delayLdRatePct') / 100) * weeks;
    const ldDelay = Math.min(ldDelayRaw, epcContract * (num('delayLdCapPct') / 100));
    const delayNet = delayLoss - ldDelay;
    $('delayBody').innerHTML = [
      row('dlContract', '도급액(① 공사비)', 'EPC 단가 × 설비용량', won(epcContract)),
      row('dlLoss', '지연에 따른 매출 손실', `${fmt(weeks, 0)}주 = ${fmt(weeks * 7, 0)}일 · 1년차 발전매출 일할`, won(delayLoss), delayLoss > 0 ? 'over' : ''),
      row('dlLd', '수령 LD', `요율 ${fmt(num('delayLdRatePct'), 2)}%/주 × ${fmt(weeks, 0)}주 = ${fmt(num('delayLdRatePct') * weeks, 2)}%${ldDelayRaw > ldDelay ? ` → 상한 ${fmt(num('delayLdCapPct'), 0)}%에서 제한` : ''}`, won(ldDelay), ldDelay > 0 ? 'best' : ''),
      row('dlNet', 'LD 반영 후 순손실', '매출 손실 − 수령 LD', won(delayNet), delayNet > 0 ? 'over' : 'best'),
    ].join('');

    // ── 하자보수 보증 효과 — 보증기간 중 고정 O&M 경감(모델에 이미 반영) ──
    const fixedOpex = sum(model.opex.items.map((it) => it.value)); // 억원/년(1년차 기준)
    const dw = model.opex.defectWarranty || { turbineYears: 0, turbinePct: 0, epcYears: 0, epcPct: 0 };
    const saveTurbine = fixedOpex * rate(dw.turbinePct) * dw.turbineYears;
    const saveEpc = fixedOpex * rate(dw.epcPct) * dw.epcYears;
    const noWarranty = cloneModel(model);
    noWarranty.opex = { ...noWarranty.opex, defectWarranty: null };
    const solvedNo = solveCurrentModel(cloneModel(noWarranty));
    noWarranty.revenue.bidPricePerKWh = Number.isFinite(price) ? price : 0;
    const rNo = analyzeEss(noWarranty);
    const withW = cloneModel(model);
    withW.revenue.bidPricePerKWh = Number.isFinite(price) ? price : 0;
    const rYes = analyzeEss(withW);
    const irrTxt = (r) => (r && !(r.errors && r.errors.length) ? `${fmt(r.projectIrr * 100, 2)}%` : '산정 불가');
    $('defectBody').innerHTML = [
      row('dfFixed', '고정 O&amp;M (1년차)', 'E. OPEX 항목 합계', won(fixedOpex)),
      row('dfTurbine', '터빈 보증 경감', `${fmt(dw.turbineYears, 0)}년 × ${fmt(dw.turbinePct, 0)}% <span class="wg wg-t">터빈사</span>`, `${won(fixedOpex * rate(dw.turbinePct))}/년 · 누적 ${won(saveTurbine)}`, saveTurbine > 0 ? 'best' : ''),
      row('dfEpc', 'BOP 보증 경감', `${fmt(dw.epcYears, 0)}년 × ${fmt(dw.epcPct, 0)}% <span class="wg wg-e">EPC</span>`, `${won(fixedOpex * rate(dw.epcPct))}/년 · 누적 ${won(saveEpc)}`, saveEpc > 0 ? 'best' : ''),
      row('dfIrr', '하자보수 보증 반영 / 미반영', '같은 수령단가 기준 P-IRR', `${irrTxt(rYes)} / ${irrTxt(rNo)}`),
      row('dfPrice', '적정 수령단가 차이', '보증 미반영 − 반영', solvedNo.ok && lastSolved && lastSolved.ok ? `${fmt(solvedNo.price - lastSolved.price, 2)} 원/kWh (${fmt(solvedNo.price, 2)} ← ${fmt(lastSolved.price, 2)})` : '—'),
    ].join('');
