  // ---------- 풍황 확률수준(P50 · P75 · P90) ----------
  // 보고서의 순이용률을 그대로 받아 선택한 확률수준을 1년차 순이용률에 반영한다.
  // P50을 채우면 (선택값 ÷ P50) 비율을 모든 터빈에 같이 곱해 터빈별 차이를 유지하고,
  // P50이 비어 있으면 지금 고른 터빈의 기준값(터빈사 제공값 또는 모델 계산값)을 P50으로 본다.
  const P_LEVELS = [['p50', 'P50'], ['p75', 'P75'], ['p90', 'P90']];
  const pLevelValue = (key) => {
    const el = $(`${key}RatePct`);
    if (!el || String(el.value).trim() === '') return null;
    const v = Number(el.value);
    return Number.isFinite(v) && v > 0 ? v : null;
  };
  // 확률수준 환산의 기준값 — P50 입력이 있으면 그 값, 없으면 지금 고른 터빈의 기준 순이용률.
  const pLevelBase = () => pLevelValue('p50') ?? turbineDerivedBase($('turbineMaker').value).y1Base;
  function pLevelFactor() {
    const mode = $('pLevelApplied').value;
    if (mode === 'none') return 1;
    const pick = pLevelValue(mode);
    if (pick == null) return 1;
    const base = pLevelBase();
    return base > 0 ? pick / base : 1;
  }

  // 확률수준을 곱하기 전까지의 값 — 모델 계산(y1Model)과 터빈사 제공값(y1Supp), 실제로 쓰는 값(y1Base).
  function turbineDerivedBase(id) {
    const sp = turbineSpec(id);
    const specificPower = sp.rotor > 0 ? sp.mw * 1e6 / (Math.PI * (sp.rotor / 2) ** 2) : 0;
    const curveF = specificPower > 0 ? ((+$('refSpecificPower').value || specificPower) / specificPower) ** (+$('powerCurveExp').value || 0) : 1;
    const losses = (1 - rate(+$('wakeLossPct').value || 0)) * (1 - rate(+$('electricalLossPct').value || 0));
    // 보증 주체별 곱 — 터빈사(출력곡선 보증 이행률 × 기술적 가용률) × EPC·O&M(BOP·계통 가용률)
    const warranty = rate(+$('powerCurveWarrantyPct').value || 0) * rate(sp.avail) * rate(+$('bopAvailPct').value || 0);
    const y1Model = Math.round((+$('siteBaseRatePct').value || 0) * curveF * warranty * losses * 100) / 100;
    const epcAuto = Math.round((sp.price + (+$('bosUnitPrice').value || 0)) * 10) / 10;
    const y1Supp = turbineOverride(id, 'y1ov'), epcSupp = turbineOverride(id, 'epcov');
    return { ...sp, specificPower, curveF, y1Model, y1Auto: y1Model, y1Supp, y1Base: y1Supp ?? y1Model, epcAuto, epcSupp, epc: epcSupp ?? epcAuto };
  }
  // 최종 파생값 = 터빈 기준값 × 확률수준 계수.
  // y1Scale — 최종값이 모델 계산값의 몇 배인지. 보증·LD 탭이 단계별 분해를 이 배율로 맞춰,
  // 보증 기준 이용률이 화면의 1년차 순이용률과 어긋나지 않게 한다.
  function turbineDerived(id) {
    const b = turbineDerivedBase(id);
    const pFactor = pLevelFactor();
    const y1 = Math.round(b.y1Base * pFactor * 100) / 100;
    return { ...b, pFactor, y1, y1Ov: b.y1Supp != null, epcOv: b.epcSupp != null,
      y1Scale: b.y1Model > 0 ? y1 / b.y1Model : 1 };
  }

    const deg = rate(+$('degradationPct').value || 0);

    } else warn.className = 'warn';
  }
