  // ---------- 발전량 구성 — 보증 주체별 단계 표 ----------
  // 해상풍력 계약 구조: 터빈사(TSA/LTSA)가 출력곡선·기술적 가용률을, EPC(BOP)가 전기·송전 손실과
  // BOP·계통 가용률을 보증한다. 후류 손실·풍황은 개발사 몫이라 보증 대상이 아니다.
  function renderGenerationBuildup() {
    const d = turbineDerived($('turbineMaker').value), tb = turbineById($('turbineMaker').value) || {};
    const base = +$('siteBaseRatePct').value || 0;
    const pcw = +$('powerCurveWarrantyPct').value || 0, bop = +$('bopAvailPct').value || 0;
    const wake = +$('wakeLossPct').value || 0, elec = +$('electricalLossPct').value || 0;
    let acc = base;
    const step = (label, owner, text, value) => {
      acc = value;
      return `<tr><th>${label}${owner} <span class="sub">${text}</span></th><td>${fmt(acc, 2)}%</td></tr>`;
    };
    const rows = [
      `<tr><th>부지 총이용률 <span class="sub">기준 터빈(비출력 ${fmt(+$('refSpecificPower').value || 0, 0)}W/m²) 기준 · 풍황</span></th><td>${fmt(base, 2)}%</td></tr>`,
      step('출력곡선 보정', '<span class="wg wg-t">터빈사</span>', `× ${fmt(d.curveF, 4)} (비출력 ${fmt(d.specificPower, 0)}W/m²)`, acc * d.curveF),
      step('출력곡선 보증 이행률', '<span class="wg wg-t">터빈사</span>', `× ${fmt(pcw, 1)}%`, acc * rate(pcw)),
      step('기술적 가용률 보증', '<span class="wg wg-t">터빈사</span>', `× ${fmt(d.avail, 1)}% (${escapeHtml(tb.maker || '')} ${escapeHtml(tb.model || '')})`, acc * rate(d.avail)),
      step('BOP·계통 가용률 보증', '<span class="wg wg-e">EPC·O&amp;M</span>', `× ${fmt(bop, 1)}%`, acc * rate(bop)),
      step('후류(웨이크) 손실', '<span class="wg wg-d">개발사</span>', `× (1 − ${fmt(wake, 1)}%)`, acc * (1 - rate(wake))),
      step('전기·송전 손실', '<span class="wg wg-e">EPC</span>', `× (1 − ${fmt(elec, 1)}%)`, acc * (1 - rate(elec))),
    ];
    rows.push(`<tr class="strong"><th>= 1년차 순이용률 <span class="sub">출력제어·계통 제약 ${fmt(+$('curtailmentPct').value || 0, 1)}%는 발전량 단계에서 따로 차감</span></th><td>${fmt(d.y1, 2)}%</td></tr>`);
    $('genBuildupBody').innerHTML = rows.join('');
  }

  // ---------- 보증·LD 탭 — 보증 미달 시 귀책별 손실과 LD 보전(탭이 보일 때만 역산) ----------
  function renderWarranty() {
    const model = readModel(), rv = model.revenue;
    const d = turbineDerived($('turbineMaker').value);
    const mw = rv.contractCapacityMW || 0, curt = 1 - rate(rv.curtailmentPct || 0);
    const price = lastSolved && lastSolved.ok ? lastSolved.price : NaN;
    const base = +$('siteBaseRatePct').value || 0, wake = 1 - rate(+$('wakeLossPct').value || 0);
    const num = (id) => +$(id).value || 0;
    // 보증 조합과 실적 조합 — 한 항목씩 바꿔 귀책을 나눈다.
    const cfOf = (pcw, avail, bop, elecLoss) => base * d.curveF * rate(pcw) * rate(avail) * rate(bop) * wake * (1 - rate(elecLoss));
    const gEl = num('electricalLossPct'), aEl = num('actElectricalLossPct');
    const cfGuar = cfOf(num('powerCurveWarrantyPct'), d.avail, num('bopAvailPct'), gEl);
    const cfTurbine = cfOf(num('actPowerCurvePct'), num('actTurbineAvailPct'), num('bopAvailPct'), gEl); // 터빈 실적만 반영
    const cfAct = cfOf(num('actPowerCurvePct'), num('actTurbineAvailPct'), num('actBopAvailPct'), aEl); // 터빈 + EPC 실적
    const genOf = (cf) => mw * 24 * 365 * rate(cf) * curt; // MWh/년
    const moneyOf = (genMWh) => (Number.isFinite(price) ? (genMWh * 1000 * price) / HUNDRED_MILLION : NaN); // 억원
    const lossTurbine = moneyOf(genOf(cfGuar) - genOf(cfTurbine));
    const lossEpc = moneyOf(genOf(cfTurbine) - genOf(cfAct));
    const turbineContract = d.price * mw, bopContract = (+$('bosUnitPrice').value || 0) * mw; // 억원
    const capT = (num('ldTurbineCapPct') / 100) * turbineContract, capE = (num('ldEpcCapPct') / 100) * bopContract;
    const ldT = Math.min(Math.max(0, lossTurbine) * (num('ldTurbinePct') / 100), capT);
    const ldE = Math.min(Math.max(0, lossEpc) * (num('ldEpcPct') / 100), capE);
    const lossTotal = lossTurbine + lossEpc, ldTotal = ldT + ldE;
    const sameAsWarranty = Math.abs(cfAct - cfGuar) < 0.005;
    $('warrantyBaseNote').textContent = `보증값 — 출력곡선 ${fmt(num('powerCurveWarrantyPct'), 1)}% · 가용률 ${fmt(d.avail, 1)}% · BOP ${fmt(num('bopAvailPct'), 1)}% · 전기손실 ${fmt(gEl, 1)}%`;
    const st = $('warrantyState');
    st.textContent = sameAsWarranty ? '● 실적 = 보증값 (손실 없음)' : `○ 실적이 보증 대비 순이용률 ${fmt(cfGuar - cfAct, 2)}%p 낮음`;
    st.className = sameAsWarranty ? 'module-state on' : 'module-state';
    const won = (v) => (Number.isFinite(v) ? `${fmt(v, 2)} 억원` : '—');
    const row = (id, th, sub, text, cls = '') => `<tr><th>${th} <span class="sub">${sub}</span></th><td id="${id}" class="${cls}">${text}</td></tr>`;
    $('warrantyBody').innerHTML = [
      row('wrCfGuar', '보증 기준 순이용률 · 발전량', '전체 연도 기준', `${fmt(cfGuar, 2)}% · ${fmt(genOf(cfGuar) / 1000, 1)} GWh`),
      row('wrCfAct', '실적 기준 순이용률 · 발전량', '터빈 + EPC 실적 반영', `${fmt(cfAct, 2)}% · ${fmt(genOf(cfAct) / 1000, 1)} GWh`, sameAsWarranty ? '' : 'over'),
      row('wrLossT', '터빈사 귀책 손실매출', '출력곡선 · 기술적 가용률 미달', won(lossTurbine), lossTurbine > 0 ? 'over' : ''),
      row('wrLossE', 'EPC사 귀책 손실매출', 'BOP·계통 가용률 · 전기손실 초과', won(lossEpc), lossEpc > 0 ? 'over' : ''),
      row('wrLossTotal', '손실매출 합계', '연간', won(lossTotal), lossTotal > 0 ? 'over' : ''),
      row('wrLdT', '터빈사 LD 보전', `보전율 ${fmt(num('ldTurbinePct'), 0)}% · 상한 ${won(capT)}(터빈 계약 ${won(turbineContract)}의 ${fmt(num('ldTurbineCapPct'), 0)}%)`, won(ldT), ldT > 0 ? 'best' : ''),
      row('wrLdE', 'EPC사 LD 보전', `보전율 ${fmt(num('ldEpcPct'), 0)}% · 상한 ${won(capE)}(BOP 계약 ${won(bopContract)}의 ${fmt(num('ldEpcCapPct'), 0)}%)`, won(ldE), ldE > 0 ? 'best' : ''),
      row('wrNet', 'LD 보전 후 순손실', '손실매출 − LD 보전 · 연간', won(lossTotal - ldTotal), lossTotal - ldTotal > 0 ? 'over' : 'best'),
    ].join('');

    // 시나리오 비교 — 보증 기준 / 실적 / 실적 + LD 보전(보전액을 같은 단가로 발전량에 환산)
    const deg = rate(+$('degradationPct').value || 0);
    const withCf = (cf) => {
      const m = cloneModel(model);
      m.revenue.operatingRatesPct = rv.operatingRatesPct.map((_, i) => Math.round(cf * (1 - deg) ** i * 100) / 100);
      return m;
    };
    const cfComp = Number.isFinite(price) && price > 0 && mw > 0
      ? cfAct + (ldTotal * HUNDRED_MILLION) / (price * 1000 * mw * 24 * 365 * curt) * 100 : cfAct;
    const scenarios = [
      { label: '보증 기준', sub: '계약상 보증값 그대로', cf: cfGuar },
      { label: '실적 (보전 전)', sub: '터빈 + EPC 실적', cf: cfAct },
      { label: '실적 + LD 보전', sub: '보전액을 발전량으로 환산', cf: cfComp },
    ];
    $('warrantyTableBody').innerHTML = scenarios.map((s) => {
      const solved = solveCurrentModel(withCf(s.cf));
      const atPrice = withCf(s.cf);
      atPrice.revenue.bidPricePerKWh = Number.isFinite(price) ? price : 0;
      const analyzed = analyzeEss(atPrice);
      const ok = !(analyzed.errors && analyzed.errors.length);
      return `
      <tr>
        <th class="mk-name">${s.label}<span class="sub">${s.sub}</span></th>
        <td>${fmt(s.cf, 2)}%</td>
        <td>${fmt(genOf(s.cf) / 1000, 1)}</td>
        <td>${Number.isFinite(price) ? fmt(moneyOf(genOf(s.cf)), 2) : '—'}</td>
        <td>${ok ? `${fmt(analyzed.projectIrr * 100, 2)}%` : '산정 불가'}</td>
        <td>${solved.ok ? fmt(solved.price, 2) : '산정 불가'}</td>
      </tr>`;
    }).join('');
  }

//@@ CAVEAT
              <li>C. 발전매출 — <strong>보증 구조(터빈사 / EPC사)</strong>: 출력곡선 보증 이행률 100% · BOP·계통 가용률 99%는 예시값이며, 기술적 가용률은 터빈 목록의 사별 값(94~97%)을 씁니다. 실제 계약은 보증 기준(시간 기준 가용률 / 발전량 기준 가용률), 측정·검증 방식(파워커브 시험), 보증 기간(하자보수 2~5년 · LTSA 5~15년), LD 산정식과 연간·총액 상한, 초과 성과 공유가 회사·계약마다 다릅니다. "보증·LD" 탭의 LD 보전율 100% · 연간 상한 10%도 예시이며, <strong>준공지연 LD와 하자보수 보증은 반영하지 않았습니다</strong></li>
