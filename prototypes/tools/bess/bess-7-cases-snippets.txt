    const ratesOf = (v) => (v > 0 ? Array.from({ length: n }, () => v) : rv.operatingRatesPct.slice());
    const set = (id, text, cls = '') => { const el = $(id); el.textContent = text; el.className = cls; };
    const pct2 = (v) => (Number.isFinite(v) ? `${fmt(v * 100, 2)}%` : '—');
    const good = (r) => r && !(r.errors && r.errors.length);

    // ① 회차 — 우리 사업(현재 페이지)을 그 회차 낙찰가 수준으로 낙찰받았다고 보고 계산
    const roundPirrs = [];
    let roundMet = 0;
    ROUND_LIST.forEach((rd) => {
      const val = (fld) => { const v = $(`rd-${rd.id}-${fld}`).value; return v === '' ? NaN : +v; };
      const price = val('price') > 0 ? val('price') : solvedPrice;
      if (!Number.isFinite(price)) { ['rev', 'pirr', 'eirr', 'profit', 'gap'].forEach((f) => set(`rd-${rd.id}-${f}`, '—')); return; }
      const m = cloneModel(base);
      m.revenue.operatingRatesPct = ratesOf(val('rate'));
      m.revenue.bidPricePerKWh = price;
      const r = analyzeEss(m);
      if (!good(r)) { ['rev', 'pirr', 'eirr', 'profit', 'gap'].forEach((f) => set(`rd-${rd.id}-${f}`, '산정 불가')); return; }
      set(`rd-${rd.id}-rev`, fmt(r.annualRevenue, 1));
      set(`rd-${rd.id}-pirr`, pct2(r.projectIrr));
      set(`rd-${rd.id}-eirr`, pct2(r.equityIrr));
      set(`rd-${rd.id}-profit`, fmt(r.companyProfitPreTax, 1));
      const g = irrGapOf(r);
      if (g >= 0) roundMet++;
      set(`rd-${rd.id}-gap`, Number.isFinite(g) ? `${g >= 0 ? '+' : '−'}${fmt(Math.abs(g) * 100, 2)}%p` : '—', g >= 0 ? 'best' : 'over');
      roundPirrs.push(r.projectIrr);
    });

    // ② 선정 사업지 — 계약용량을 사례 값으로 바꾸고(사업비·운영비는 용량 비율) 목표 IRR 적정 단가를 역산
    const needs = [];
    let fitCount = 0;
    CASE_LIST.forEach((c) => {
      const rd = roundById(c.round);
      const val = (fld) => { const v = $(`cs-${c.id}-${fld}`).value; return v === '' ? NaN : +v; };
      const mw = val('mw') > 0 ? val('mw') : curMW, ratio = mw / curMW;
      const price = val('price') > 0 ? val('price') : solvedPrice;
      const build = (p) => {
        const m = cloneModel(base);
        m.revenue.contractCapacityMW = mw;
        m.revenue.operatingRatesPct = ratesOf(val('rate'));
        m.capex = { ...m.capex, items: m.capex.items.map((it) => ({ ...it, value: it.value * ratio })) };
        m.opex = { ...m.opex, items: m.opex.items.map((it) => ({ ...it, value: it.value * ratio })) };
        m.finance = { ...m.finance, fundingMode: 'ratio' };
        m.revenue.bidPricePerKWh = p;
        return m;
      };
      const sv = solveBidPriceForTarget(build(0), rate(+$('targetIrrPct').value || 0), $('irrKind').value);
      const r = Number.isFinite(price) ? analyzeEss(build(price)) : null;
      set(`cs-${c.id}-inv`, good(r) ? fmt(r.totalInvestment, 0) : (sv.ok ? fmt(sv.result.totalInvestment, 0) : '산정 불가'));
      set(`cs-${c.id}-need`, sv.ok ? `${fmt(sv.price, 2)}원` : '500원으로도 미달', sv.ok ? '' : 'over');
      if (sv.ok) needs.push(sv.price);
      set(`cs-${c.id}-pirr`, good(r) ? pct2(r.projectIrr) : '—');
      set(`cs-${c.id}-profit`, good(r) ? fmt(r.companyProfitPreTax, 1) : '—');
      if (!good(r) || !sv.ok) { set(`cs-${c.id}-fit`, '—'); return; }
      const gap = irrGapOf(r), fits = gap >= 0;
      if (fits) fitCount++;
      set(`cs-${c.id}-fit`, fits ? `달성 (${fmt(price - sv.price, 2)}원 여유)` : `${fmt(sv.price - price, 2)}원 부족`, fits ? 'best' : 'over');
    });

    const spread = (arr, f) => {
      const s = [...arr].sort((a, b) => a - b);
      return s.length ? { lo: f(s[0]), hi: f(s[s.length - 1]), med: f(s[Math.floor((s.length - 1) / 2)]) } : null;
    };
    const rp = spread(roundPirrs, pct2), np = spread(needs, (v) => `${fmt(v, 2)}원`);
    $('caseSummary').innerHTML = [
      rp ? `<strong>① 회차 낙찰가 수준으로 우리 사업(${fmt(curMW, 0)}MW)을 낙찰받았다면 P-IRR ${rp.lo} ~ ${rp.hi}</strong> — ${irrTargetLabel()} 달성 ${roundMet}/${ROUND_LIST.length}회차` : '',
      np ? `<strong>② 선정 사업지 ${needs.length}곳의 ${irrTargetLabel()} 적정 단가 ${np.lo} ~ ${np.hi} (중앙값 ${np.med})</strong> — 적용 단가로 목표 달성 ${fitCount}곳` : '',
    ].filter(Boolean).join('<br>') || '사례를 계산하지 못했습니다.';
  }
