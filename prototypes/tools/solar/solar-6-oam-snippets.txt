  // ---------- 발전시간(h/일) ↔ 이용률 ----------
  // 태양광 O&M 계약은 대개 일평균 발전시간(h/일) 또는 연간 발전시간을 보증한다. 발전량은 그 시간에 정비례한다:
  //   연간 발전량(MWh) = 설비용량(MW) × 보증시간(h/일) × 365일
  // 엔진은 이용률로 발전량을 계산하므로(= 용량 × 24h × 365 × 이용률 × (1 − 손실률)), 보증시간은 계량기 기준
  // 실발전으로 보고 손실률을 되돌려 이용률로 환산한다: 이용률 = 시간 ÷ 24 ÷ (1 − 손실률).
  const DAYS_PER_YEAR = 365;
  const availabilityFactor = () => 1 - rate(Math.min(100, Math.max(0, +$('curtailmentPct').value || 0)));
  const hoursFromRatePct = (ratePct) => rate(ratePct) * 24 * availabilityFactor();
  const ratePctFromHours = (hours) => (availabilityFactor() > 0 ? (hours / 24 / availabilityFactor()) * 100 : 0);
  const oamGuaranteeHoursPerDay = () => {
    const v = Math.max(0, +$('oamGuaranteeHours').value || 0);
    return $('oamHoursUnit').value === 'perYear' ? v / DAYS_PER_YEAR : v;
  };
  // 1년차 발전시간 칸을 지금 이용률·손실률에 맞춘다(그 칸을 직접 고치는 중이면 덮어쓰지 않는다 — 반대 방향은 리스너가 처리).
  function syncGenerationHours() {
    if (document.activeElement === $('year1HoursPerDay')) return;
    $('year1HoursPerDay').value = (Math.round(hoursFromRatePct(+$('year1RatePct').value || 0) * 100) / 100).toFixed(2);
  }

  // ---------- O&M 보증 탭 — 보증시간 기준 발전량·발전매출과 보증시간별 비교(탭이 보일 때만 역산) ----------
  function renderOamGuarantee() {
    const base = readModel(), rv = base.revenue;
    const mw = rv.contractCapacityMW || 0, guaranteed = oamGuaranteeHoursPerDay();
    const current = hoursFromRatePct(rv.operatingRatesPct[0] || 0);
    const genOf = (hours) => mw * hours * DAYS_PER_YEAR; // MWh/년 (전체 연도 기준)
    const price = lastSolved && lastSolved.ok ? lastSolved.price : NaN; // 원/kWh — 현재 산정된 수령단가
    const moneyOf = (genMWh) => (Number.isFinite(price) ? (genMWh * 1000 * price) / HUNDRED_MILLION : NaN); // 억원
    const cover = Math.min(100, Math.max(0, +$('oamCoverPct').value || 0)) / 100;
    const gapGen = genOf(guaranteed) - genOf(current), gapMoney = moneyOf(genOf(guaranteed)) - moneyOf(genOf(current));
    $('oamHoursUnitLabel').textContent = $('oamHoursUnit').value === 'perYear' ? 'h/년' : 'h/일';
    const near = Math.abs(guaranteed - current) < 0.005;
    const state = $('oamState');
    state.textContent = near ? `● 보증시간 ${fmt(guaranteed, 2)}h/일로 발전량 적용 중`
      : `○ 현재 발전량은 ${fmt(current, 2)}h/일 (보증 ${fmt(guaranteed, 2)}h/일 대비 ${gapGen >= 0 ? '−' : '+'}${fmt(Math.abs(gapGen), 0)}MWh/년)`;
    state.className = near ? 'module-state on' : 'module-state';

    // 계약기간 누적 — 보증시간은 매년 같고, 현재 가정은 열화율만큼 줄어든다(계약기간과 운영기간 중 짧은 쪽).
    const years = Math.max(0, Math.min(Math.round(rv.contractYears || 0), rv.operatingRatesPct.length));
    let cumGap = 0;
    for (let i = 0; i < years; i++) cumGap += genOf(guaranteed) - genOf(hoursFromRatePct(rv.operatingRatesPct[i] || 0));
    const row = (id, th, sub, text, cls = '') => `<tr><th>${th} <span class="sub">${sub}</span></th><td id="${id}" class="${cls}">${text}</td></tr>`;
    $('oamBody').innerHTML = [
      row('oamHours', '보증 발전시간', '일평균 · 연간', `${fmt(guaranteed, 2)} h/일 · ${fmt(guaranteed * DAYS_PER_YEAR, 0)} h/년`),
      row('oamRate', '이용률 환산', '= 시간 ÷ 24h ÷ (1 − 손실률)', `${fmt(ratePctFromHours(guaranteed), 2)}% (현재 ${fmt(rv.operatingRatesPct[0] || 0, 2)}%)`),
      row('oamGen', '보증 기준 연간 발전량', '= 용량 × 보증시간 × 365일', `${fmt(genOf(guaranteed), 0)} MWh`),
      row('oamGenNow', '현재 가정 연간 발전량', `${fmt(current, 2)} h/일 기준`, `${fmt(genOf(current), 0)} MWh`),
      row('oamGenGap', '발전량 차이', '보증 − 현재', `${gapGen >= 0 ? '+' : ''}${fmt(gapGen, 0)} MWh${genOf(current) > 0 ? ` (${gapGen >= 0 ? '+' : ''}${fmt((gapGen / genOf(current)) * 100, 1)}%)` : ''}`, gapGen >= 0 ? 'best' : 'over'),
      row('oamPrice', '적용 수령단가', '현재 산정값', Number.isFinite(price) ? `${fmt(price, 2)} 원/kWh` : '— (산정 불가)'),
      row('oamMoney', '보증 기준 연간 발전매출', '= 발전량 × 수령단가', Number.isFinite(price) ? `${fmt(moneyOf(genOf(guaranteed)), 2)} 억원` : '—'),
      row('oamMoneyNow', '현재 가정 연간 발전매출', `${fmt(current, 2)} h/일 기준`, Number.isFinite(price) ? `${fmt(moneyOf(genOf(current)), 2)} 억원` : '—'),
      row('oamMoneyGap', '발전매출 차이', '보증 − 현재 · 연간', Number.isFinite(price) ? `${gapMoney >= 0 ? '+' : ''}${fmt(gapMoney, 2)} 억원` : '—', gapMoney >= 0 ? 'best' : 'over'),
      row('oamCover', '미달분 보전액', `보증 미달 × 수령단가 × 보전율 ${fmt(cover * 100, 0)}%`, Number.isFinite(price) ? `${fmt(Math.max(0, gapMoney) * cover, 2)} 억원/년` : '—'),
      row('oamCum', `계약기간(${fmt(years, 0)}년) 누적 발전량 차이`, '열화율 반영 · 보증시간은 매년 동일 가정', `${cumGap >= 0 ? '+' : ''}${fmt(cumGap, 0)} MWh${Number.isFinite(price) ? ` · ${cumGap >= 0 ? '+' : ''}${fmt(moneyOf(cumGap), 1)} 억원` : ''}`),
    ].join('');

    // 보증시간별 비교 — 시간마다 1년차 이용률을 환산해 넣고(열화율은 그대로) 목표 달성 단가를 다시 푼다.
    const deg = rate(+$('degradationPct').value || 0);
    const list = [...new Set([3.0, 3.2, 3.4, 3.6, 3.8, 4.0, 4.2, Math.round(guaranteed * 100) / 100, Math.round(current * 100) / 100])]
      .filter((h) => h > 0).sort((a, b) => a - b);
    const withRates = (hours) => {
      const m = cloneModel(base), r0 = ratePctFromHours(hours);
      m.revenue.operatingRatesPct = rv.operatingRatesPct.map((_, i) => Math.round(r0 * (1 - deg) ** i * 100) / 100);
      return m;
    };
    const rows = list.map((hours) => {
      const solved = solveCurrentModel(withRates(hours));
      const atPrice = withRates(hours);
      atPrice.revenue.bidPricePerKWh = Number.isFinite(price) ? price : 0;
      const analyzed = analyzeEss(atPrice);
      const ok = !(analyzed.errors && analyzed.errors.length);
      return { hours, gen: genOf(hours), money: moneyOf(genOf(hours)), irr: ok ? analyzed.projectIrr : NaN, solve: solved.ok ? solved.price : NaN };
    });
    const bestIrr = Math.max(...rows.map((r) => (Number.isFinite(r.irr) ? r.irr : -Infinity)));
    $('oamTableBody').innerHTML = rows.map((r) => {
      const isNow = Math.abs(r.hours - current) < 0.005, isGuarantee = Math.abs(r.hours - guaranteed) < 0.005;
      const tag = isGuarantee ? '<span class="mk-origin domestic">보증</span>' : isNow ? '<span class="mk-origin china">현재</span>' : '';
      return `
      <tr${isNow ? ' class="is-picked"' : ''}>
        <th class="mk-name">${tag}${fmt(r.hours, 2)} h/일<span class="sub">${fmt(r.hours * DAYS_PER_YEAR, 0)} h/년</span></th>
        <td>${fmt(ratePctFromHours(r.hours), 2)}%</td>
        <td>${fmt(r.gen, 0)}</td>
        <td>${Number.isFinite(r.money) ? fmt(r.money, 2) : '—'}</td>
        <td class="${Number.isFinite(r.irr) && r.irr === bestIrr ? 'best' : ''}">${Number.isFinite(r.irr) ? `${fmt(r.irr * 100, 2)}%` : '산정 불가'}</td>
        <td>${Number.isFinite(r.solve) ? fmt(r.solve, 2) : '산정 불가'}</td>
      </tr>`;
    }).join('');
  }
