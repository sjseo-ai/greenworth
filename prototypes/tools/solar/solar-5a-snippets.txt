  // 새 모듈(또는 새 제조사)은 MODULE_CATALOG에 항목 하나를 더하면 C. 발전매출의 선택 목록·비교표·탄소검증 표·스펙표·근거 목록에 자동 반영된다.
  // 필드: id(영문·숫자·_ 고유값) · origin('domestic' | 'china') · name(제조사) · model(대표 모델) · specLabel(스펙 확인 수준, 짧게)
  //       registered(탄소검증 등록명·공장) · dist(검증 제품 등급 분포 문구) · price(모듈 단가 원/W, 예시) · gamma(온도계수 %/℃)
  //       bif(양면율 %) · deg1(1년차 열화 %) · degA(연간 열화 %/년) · products([{ v, label, co2 }] — 첫 항목이 기본 선택, co2 0 = 미검증)
  //       source(스펙 근거 문장). 화면의 "새 모듈 등록"으로 추가한 모듈도 같은 모양으로 저장·내보낸다.
  const MODULE_CATALOG_UPDATED = '2026-09-14';
  const MODULE_CATALOG = __CATALOG__;
  // 사용자 추가 모듈 — 이 브라우저(localStorage)에 저장하고, 시나리오 저장·내보내기에도 담는다.
  const CUSTOM_MODULE_KEY = 'solarCustomModules.v1';
  const MODULE_ID_RE = /^[A-Za-z0-9_]+$/; // 표 칸 id(mk-{id}-price)에 들어가므로 하이픈 없이
  function validModule(t) { return !!(t && t.name && t.model && +t.price > 0); }
  // 빈 필드를 채운다 — 탄소검증 제품 목록이 없으면 배출량(co2)으로 한 항목 + 미검증 항목을 만든다.
  function normalizeModule(t) {
    const num = (v, d) => (v === '' || v == null || !Number.isFinite(+v) ? d : +v);
    const co2 = num(t.co2, 0);
    const products = Array.isArray(t.products) && t.products.length
      ? t.products.map((p) => ({ v: String(p.v), label: String(p.label || p.v), co2: num(p.co2, 0) }))
      : [...(co2 > 0 ? [{ v: 'verified', label: `검증 배출량 · ${co2.toFixed(2)}`, co2 }] : []), { v: 'none', label: '미검증 제품 사용', co2: 0 }];
    const out = { ...t, id: String(t.id), origin: t.origin === 'domestic' ? 'domestic' : 'china', name: String(t.name).trim(), model: String(t.model).trim(),
      price: num(t.price, 0), gamma: num(t.gamma, -0.3), bif: num(t.bif, 80), deg1: num(t.deg1, 1), degA: num(t.degA, 0.4), products };
    delete out.co2; delete out.custom;
    return out;
  }
  function loadCustomModules() {
    try {
      const v = JSON.parse(localStorage.getItem(CUSTOM_MODULE_KEY) || '[]');
      return Array.isArray(v) ? v.filter((t) => validModule(t) && MODULE_ID_RE.test(String(t.id))).map(normalizeModule) : [];
    } catch (e) { return []; }
  }
  function saveCustomModules() {
    try { localStorage.setItem(CUSTOM_MODULE_KEY, JSON.stringify(customModules)); return true; } catch (e) { return false; }
  }
  let customModules = loadCustomModules();
  const newModuleId = () => 'm' + Date.now().toString(36) + Math.floor(Math.random() * 46656).toString(36);
  const allModules = () => [...MODULE_CATALOG, ...customModules.map((t) => ({ ...t, custom: true }))];
  const makerById = (id) => allModules().find((m) => m.id === id);
  const productsOf = (id) => (makerById(id) || {}).products || [];

  // 기자재 원산지가 EPC 단가와 탄소 등급을 동시에 움직이고, 그에 따라 판매 구조까지 달라지는 두 조합.
  // 조합을 따로 저장하지 않고 "지금 입력값이 어느 조합과 일치하는지"로 판정한다(시나리오 저장·복원과 자동으로 맞는다).

  // 이미 있던 표 칸의 값(이번 세션에 고친 값)과 선택 모듈사는 유지한다. 칸의 HTML value·selected는 정의값이라 "기본값으로 되돌리기"가 그 값으로 돌아간다.
  function renderModuleUI() {
    const list = allModules(), keep = {}, h = escapeHtml;
    document.querySelectorAll('#moduleCarbonBody [id^="mk-"], #moduleSpecBody [id^="mk-"]').forEach((el) => { keep[el.id] = el.value; });
    const sel = $('moduleMaker'), sec = $('secondaryMaker'), prevSel = sel.value, prevSec = sec.value;
    const originLabel = (m) => (m.origin === 'domestic' ? '국산' : '중국산');
    const opt = (m) => `<option value="${h(m.id)}"${m.id === 'hanwha' ? ' selected' : ''}>${h(m.name)} · ${h(m.model)}${m.custom ? ' (사용자 추가)' : ''}</option>`;
    sel.innerHTML = `<optgroup label="국산 (탄소검증 우대 대상)">${list.filter((m) => m.origin === 'domestic').map(opt).join('')}</optgroup>`
      + `<optgroup label="중국산">${list.filter((m) => m.origin !== 'domestic').map(opt).join('')}</optgroup>`;
    sel.value = list.some((m) => m.id === prevSel) ? prevSel : 'hanwha';
    sec.innerHTML = '<option value="" selected>혼용 없음</option>' + list.map((m) => `<option value="${h(m.id)}">${h(m.name)}${m.custom ? ` · ${h(m.model)}` : ''}</option>`).join('');
    sec.value = list.some((m) => m.id === prevSec) ? prevSec : '';
    const val = (v, d) => (v === '' || v == null || !Number.isFinite(+v) ? '' : d == null ? String(+v) : (+v).toFixed(d));
    const nameCell = (m, sub) => `<th class="mk-name"><span class="mk-origin ${m.origin === 'domestic' ? 'domestic' : 'china'}">${originLabel(m)}</span>${h(m.name)}<span class="sub">${sub}</span></th>`;
    const specSub = (m) => `${h(m.model)} · ${h(m.custom ? '사용자 추가' : m.specLabel || '')}`;
    $('moduleTableBody').innerHTML = list.map((m) => {
      const id = h(m.id);
      return `
                  <tr data-mk-row="${id}">${nameCell(m, specSub(m))}
                    <td id="mk-${id}-y1">—</td><td id="mk-${id}-avg">—</td><td id="mk-${id}-epc">—</td><td id="mk-${id}-solve">—</td><td id="mk-${id}-bid">—</td>
                    <td><button type="button" class="btn btn-soft" data-mk-pick="${id}">선택</button></td>
                  </tr>`;
    }).join('');
    $('moduleCarbonBody').innerHTML = list.map((m) => {
      const id = h(m.id), p0 = m.products[0] || { co2: 0 };
      const opts = m.products.map((p, i) => `<option value="${h(p.v)}"${i === 0 ? ' selected' : ''}>${h(p.label)}</option>`).join('');
      return `
                  <tr data-mk-row="${id}">${nameCell(m, h(m.registered || (m.custom ? '사용자 추가' : '')))}
                    <td style="text-align:left;"><select id="mk-${id}-product" class="mk-product" aria-label="${h(m.name)} 탄소검증 제품">${opts}<option value="custom">직접 입력</option></select></td>
                    <td><input id="mk-${id}-co2" type="number" value="${p0.co2 > 0 ? p0.co2 : ''}" step="0.01" min="0" placeholder="미검증" aria-label="${h(m.name)} 탄소배출량"></td>
                    <td id="mk-${id}-cgrade">—</td><td id="mk-${id}-cscore">—</td><td id="mk-${id}-cprem">—</td>
                    <td>${h(m.dist || (m.custom ? '사용자 입력' : ''))}</td>
                  </tr>`;
    }).join('');
    const cell = (m, fld, step, label, d) => `<td><input id="mk-${h(m.id)}-${fld}" type="number" value="${val(m[fld], d)}" step="${step}" aria-label="${h(m.name)} ${label}"></td>`;
    $('moduleSpecBody').innerHTML = list.map((m) => `
                  <tr data-mk-row="${h(m.id)}">${nameCell(m, specSub(m))}${cell(m, 'price', 5, '모듈 단가')}${cell(m, 'gamma', 0.01, '온도계수', 2)}${cell(m, 'bif', 1, '양면율')}${cell(m, 'deg1', 0.1, '1년차 열화', 1)}${cell(m, 'degA', 0.01, '연간 열화', 2)}
                    <td>${m.custom ? `<button type="button" class="btn" data-mk-del="${h(m.id)}">삭제</button>` : ''}</td>
                  </tr>`).join('');
    document.querySelectorAll('#moduleCarbonBody [id^="mk-"], #moduleSpecBody [id^="mk-"]').forEach((el) => { if (keep[el.id] != null) el.value = keep[el.id]; });
    $('moduleSourceList').innerHTML = list.map((m) => `<li>${m.custom ? `<strong>[사용자 추가 ${h(m.addedAt || '')}]</strong> ` : ''}${h(m.source || `${m.name} ${m.model} — 근거 미입력`)}</li>`).join('');
    const dom = list.filter((m) => m.origin === 'domestic').length;
    $('moduleCountHead').textContent = `국산 ${dom}종 · 중국산 ${list.length - dom}종`;
    $('moduleCatalogInfo').textContent = `기본 목록 ${MODULE_CATALOG.length}종 (기준일 ${MODULE_CATALOG_UPDATED}) · 사용자 추가 ${customModules.length}종`;
    resultsDirty.module = true;
  }

  function removeCustomModule(id) {
    const t = customModules.find((x) => x.id === id);
    if (!t) return;
    customModules = customModules.filter((x) => x.id !== id);
    saveCustomModules();
    const wasSelected = $('moduleMaker').value === id, wasSecondary = $('secondaryMaker').value === id;
    renderModuleUI();
    if (wasSelected || (wasSecondary && moduleLinked)) applyModuleDerived();
    recalcAll();
    moduleMsg(`${t.name} ${t.model}을(를) 목록에서 뺐습니다${wasSelected ? ' — 선택 모듈사는 한화큐셀로 바뀌었습니다' : ''}${wasSecondary ? ' — 혼용 모듈사는 "혼용 없음"으로 바뀌었습니다' : ''}. 내보낸 파일이 있으면 불러오기로 되살릴 수 있습니다.`, false);
  }
  // 같은 id(또는 같은 제조사+모델)의 사용자 추가 모듈은 갱신, 새 것은 추가. 기본 목록과 같은 제조사+모델은 건너뛴다.
  function mergeModules(incoming) {
    const r = { added: 0, updated: 0, skipped: 0 };
    (Array.isArray(incoming) ? incoming : []).forEach((raw) => {
      if (!validModule(raw)) { r.skipped++; return; }
      const t = normalizeModule(raw);
      if (MODULE_CATALOG.some((x) => x.name === t.name && x.model === t.model)) { r.skipped++; return; }
      if (!MODULE_ID_RE.test(t.id) || MODULE_CATALOG.some((x) => x.id === t.id)) t.id = newModuleId();
      const i = customModules.findIndex((x) => x.id === t.id || (x.name === t.name && x.model === t.model));
      if (i >= 0) { customModules[i] = { ...customModules[i], ...t, id: customModules[i].id }; r.updated++; } else { customModules.push(t); r.added++; }
    });
    if (r.added + r.updated) saveCustomModules();
    return r;
  }

    if (!customModules.length) return moduleMsg('내보낼 사용자 추가 모듈이 없습니다.', true);
    if (!downloadsApi) return moduleMsg('이 화면에서는 다운로드 기능을 사용할 수 없습니다(Claude 앱/웹의 아티팩트 화면에서 다시 시도해 주세요).', true);
    try {
      const data = JSON.stringify({ tool: '태양광 적정 입찰가격 프로토타입 (초안)', kind: 'moduleCatalog', exportedAt: new Date().toISOString(),
        modules: customModules.map(({ custom, ...t }) => t) }, null, 2);
      await downloadsApi.save({ filename: `태양광_모듈목록_${new Date().toISOString().slice(0, 10)}.json`, data });
      moduleMsg(`사용자 추가 모듈 ${customModules.length}종을 내보냈습니다.`, false);
    } catch (e) {
      moduleMsg(e && e.code === 'declined' ? '다운로드를 취소했습니다.' : '다운로드에 실패했습니다: ' + (e && e.message ? e.message : e), true);
    }
  });
  $('importModulesBtn').addEventListener('click', () => $('importModulesFile').click());
  $('importModulesFile').addEventListener('change', async (e) => {
    const file = e.target.files && e.target.files[0];
    if (!file) return;
    try {
      const parsed = JSON.parse(await file.text());
      // 모듈 목록 파일(modules) · 시나리오 파일(inputs.customModules) · 항목 배열 모두 받는다
      const incoming = Array.isArray(parsed) ? parsed : parsed.modules || (parsed.inputs && parsed.inputs.customModules) || parsed.customModules || [];
      const r = mergeModules(incoming);
      renderModuleUI();
      recalcAll();
      moduleMsg(`불러오기: 추가 ${r.added}종 · 갱신 ${r.updated}종${r.skipped ? ` · 건너뜀 ${r.skipped}종(필수값 누락 또는 기본 목록과 같은 모델)` : ''}.`, r.added + r.updated === 0);
    } catch (err) {
      moduleMsg('JSON 파일을 읽지 못했습니다: ' + (err && err.message ? err.message : err), true);
    } finally {
      e.target.value = '';
    }
  });
