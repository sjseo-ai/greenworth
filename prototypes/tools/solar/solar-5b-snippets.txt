  // ---------- 목표 대비 여유 · 이분탐색 도우미 ----------
  // 목표(A. 산정 기준) 대비 여유 — 0 이상이면 목표 달성. IRR 기준이면 소수(0.01 = 1%p), 당사 이익 기준이면 억원.
  function targetGapOf(r) {
    if (!r || (r.errors && r.errors.length)) return NaN;
    if ($('solveMode').value === 'companyProfit') return (r[$('companyProfitKind').value] ?? NaN) - (+$('targetCompanyProfit').value || 0);
    return (r[$('irrKind').value] ?? NaN) - rate(+$('targetIrrPct').value || 0);
  }
  const isProfitTarget = () => $('solveMode').value === 'companyProfit';
  const targetMetricOf = (r) => (isProfitTarget() ? r[$('companyProfitKind').value] : r[$('irrKind').value]);
  const fmtTargetMetric = (v) => (Number.isFinite(v) ? (isProfitTarget() ? `${fmt(v, 1)}억원` : `${fmt(v * 100, 2)}%`) : '—');
  const targetLabel = () => (isProfitTarget()
    ? `목표 당사 이익(${$('companyProfitKind').selectedOptions[0].textContent.trim()}) ${fmt(+$('targetCompanyProfit').value || 0, 0)}억원`
    : `목표 ${$('irrKind').selectedOptions[0].textContent.trim()} ${fmt(+$('targetIrrPct').value || 0, 2)}%`);
  // 사례 판정용 — A. 산정 기준과 관계없이 "기준 IRR · 목표 IRR" 칸으로(당사 이익 목표는 사업 규모에 따라 달라 비교에 부적합)
  const irrGapOf = (r) => (!r || (r.errors && r.errors.length) ? NaN : (r[$('irrKind').value] ?? NaN) - rate(+$('targetIrrPct').value || 0));
  const irrTargetLabel = () => `목표 ${$('irrKind').selectedOptions[0].textContent.trim()} ${fmt(+$('targetIrrPct').value || 0, 2)}%`;
  const cloneModel = (m) => { try { return structuredClone(m); } catch (e) { return JSON.parse(JSON.stringify(m)); } };
  // ok(lo)가 참이고 ok(hi)가 거짓일 때 참인 가장 큰 값 / ok(hi)가 참일 때 참인 가장 작은 값
  function bisectMax(lo, hi, ok) { for (let i = 0; i < 32; i++) { const mid = (lo + hi) / 2; if (ok(mid)) lo = mid; else hi = mid; } return lo; }
  function bisectMin(lo, hi, ok) { for (let i = 0; i < 32; i++) { const mid = (lo + hi) / 2; if (ok(mid)) hi = mid; else lo = mid; } return hi; }

  // ---------- 상한가 진입 조건 — 한 가지만 바꿔 입찰가격을 상한가에 맞출 때의 값(목표 수익 유지, SPC 기준) ----------
  // 상한가에 입찰했을 때의 수령단가(= 상한가 + 탄소 우대가격)에서 각 항목을 바꿔 목표 여유가 0이 되는 지점을 이분탐색으로 찾는다.
  function renderEntryConditions() {
    const body = $('entryBody'), note = $('entryNote');
    const R0 = parseFloat(String($('heroBidPrice').textContent).replace(/,/g, ''));
    const base = readModel(), rv = base.revenue;
    if (rv.salesStructure !== 'fixed' || !Number.isFinite(R0) || !(rv.priceCapPerKWh > 0)) {
      note.innerHTML = '기업PPA이거나 적정 수령단가를 산정하지 못해(또는 상한가격이 없어) 상한가 진입 조건을 계산하지 않습니다.';
      body.innerHTML = '';
      return;
    }
    const cap = rv.priceCapPerKWh, prem = rv.carbonPremiumPerKWh || 0;
    const P0 = R0 - prem, gap = P0 - cap, Rcap = cap + prem;
    const mw = rv.contractCapacityMW || 1, isEpc = (it) => it.group === 'construction';
    const at = (mutate) => { const m = cloneModel(base); m.revenue.bidPricePerKWh = Rcap; if (mutate) mutate(m); return analyzeEss(m); };
    const metricCap = targetMetricOf(at());
    const row = (id, th, sub, text, value, cls = '') => `<tr><th>${th} <span class="sub">${sub}</span></th><td id="${id}" data-value="${Number.isFinite(value) ? value : ''}" class="${cls}">${text}</td></tr>`;
    const rows = [];
    if (gap <= 0) {
      note.innerHTML = `<strong>지금 조건으로 상한가 이내입니다</strong> — 써낼 입찰가격(SPC 기준) ${fmt(P0, 2)}원/kWh, 상한가 ${fmt(cap, 3)}원까지 ${fmt(-gap, 2)}원 여유. 아래는 상한가에 그대로 입찰했을 때의 상승 여지입니다.`;
      rows.push(row('entryGap', '상한가까지 여유', '상한가 − 입찰가격', `${fmt(-gap, 2)}원/kWh (상한가의 ${fmt(-gap / cap * 100, 1)}%)`, gap, 'best'));
      rows.push(row('entryTarget', '상한가에 입찰하면', targetLabel(), `${fmtTargetMetric(metricCap)} 달성`, metricCap));
      body.innerHTML = rows.join('');
      return;
    }
    note.innerHTML = `<strong>상한가를 ${fmt(gap, 2)}원/kWh 넘어 응찰할 수 없습니다</strong> — 써낼 입찰가격(SPC 기준) ${fmt(P0, 2)}원 → 상한가 ${fmt(cap, 3)}원. 아래 각 행은 나머지 조건은 그대로 두고 <strong>그 항목 하나만</strong> 바꿔 상한가에 입찰해도 목표를 지키는 값입니다.`;
    rows.push(row('entryGap', '입찰가격 인하 필요', '입찰가격 − 상한가 · SMP+1REC 기준', `${fmt(gap, 2)}원/kWh (입찰가격의 ${fmt(gap / P0 * 100, 1)}%)`, gap, 'over'));
    rows.push(row('entryTarget', '목표 수익을 낮추면', `${targetLabel()} →`, `${fmtTargetMetric(metricCap)}까지 낮추면 상한가 입찰 가능`, metricCap));
    // EPC 단가(원/W) — ① 공사비(억원) = 단가 × 설비용량(MW) ÷ 100
    const epc0 = sum(base.capex.items.filter(isEpc).map((it) => it.value)) * 100 / mw;
    const epcOk = (e) => targetGapOf(at((m) => {
      m.capex = { ...m.capex, items: [{ id: 'epcLumpSum', label: 'EPC', value: e * mw / 100, category: '기자재', group: 'construction' }, ...m.capex.items.filter((it) => !isEpc(it))] };
    })) >= 0;
    const epcReq = epcOk(0) ? bisectMax(0, epc0, epcOk) : NaN;
    rows.push(row('entryEpc', 'EPC 단가를 낮추면', '① 공사비 ÷ 설비용량 (모듈 + BOS·시공)', Number.isFinite(epcReq)
      ? `${fmt(epc0, 0)} → ${fmt(epcReq, 1)}원/W (−${fmt(epc0 - epcReq, 1)}원 · −${fmt((1 - epcReq / epc0) * 100, 1)}%)` : 'EPC를 0으로 해도 부족', epcReq, Number.isFinite(epcReq) ? '' : 'over'));
    const rates0 = rv.operatingRatesPct, kMax = 40 / Math.max(...rates0, 0.01);
    const cfOk = (k) => targetGapOf(at((m) => { m.revenue.operatingRatesPct = rates0.map((v) => v * k); })) >= 0;
    const kReq = cfOk(kMax) ? bisectMin(1, kMax, cfOk) : NaN;
    rows.push(row('entryCf', '이용률을 높이면', '전 연차를 같은 비율로 (1년차 이용률 칸)', Number.isFinite(kReq)
      ? `1년차 ${fmt(rates0[0], 2)}% → ${fmt(rates0[0] * kReq, 2)}% (+${fmt(rates0[0] * (kReq - 1), 2)}%p)` : '이용률 40%로도 부족', Number.isFinite(kReq) ? rates0[0] * kReq : NaN, Number.isFinite(kReq) ? '' : 'over'));
    const opex0 = sum(base.opex.items.map((it) => it.value));
    const opOk = (fct) => targetGapOf(at((m) => { m.opex = { ...m.opex, items: m.opex.items.map((it) => ({ ...it, value: it.value * fct })) }; })) >= 0;
    const fReq = opOk(0) ? bisectMax(0, 1, opOk) : NaN;
    rows.push(row('entryOpex', '고정 O&amp;M을 줄이면', 'E. OPEX 고정비 항목 합계', Number.isFinite(fReq)
      ? `연 ${fmt(opex0, 1)} → ${fmt(opex0 * fReq, 1)}억원 (−${fmt((1 - fReq) * 100, 1)}%)` : '고정 O&amp;M을 0으로 해도 부족', Number.isFinite(fReq) ? opex0 * fReq : NaN, Number.isFinite(fReq) ? '' : 'over'));
    // 탄소 등급 — 수령단가는 그대로, 우대가격만 커져 써낼 입찰가격이 내려간다(상위 등급 모듈의 단가 차이는 미반영)
    const table = carbonGradeTable(), grade = $('carbonGrade').value;
    ['1', '2'].filter((g) => +g < +grade && (table[g]?.premium || 0) > prem).forEach((g) => {
      const Pg = R0 - table[g].premium, fits = Pg <= cap;
      rows.push(row(`entryGrade${g}`, `탄소 ${g}등급 모듈을 쓰면`, `우대가격 ${fmt(table[g].premium, 0)}원 · 배점 ${fmt(table[g].score, 0)}점`,
        `입찰가격 ${fmt(Pg, 2)}원 — ${fits ? `상한 이내(${fmt(cap - Pg, 2)}원 여유)` : `여전히 ${fmt(Pg - cap, 2)}원 초과`}`, Pg, fits ? 'best' : 'over'));
    });
    body.innerHTML = rows.join('');
  }

  // 회차: id · label · announcedMW · receivedMW · selectedMW · receivedCount · selectedCount · cap(육지 상한가 원/kWh) · avg(낙찰 평균가 원/kWh | null)
  //       premiums({ '1': 원, '2': 원 } | null — 탄소 우대가격 도입 전은 null) · period · note · sources
  const ROUND_LIST = __ROUNDS__;
  // 대형 사업: id · name · mw · type · developer · module · epc · capexPerMW(억원/MW | null) · capexBasis(결과표 표시) · capexText
  //       epcAmount(EPC 금액 억원 | null — 총사업비가 없을 때 역산) · genGWh(연간 발전량 | null) · cf(보도 이용률 % | null)
  //       siteText · status · note · eiass(EIASS 코드·협의 이력·등록 제원 | '') · sources
  const CASE_LIST = __CASES__;
  const caseCfDefault = (c) => (c.cf != null ? c.cf : c.genGWh ? Math.round((c.genGWh / (c.mw * 8.76)) * 1000) / 10 : null);
  const roundPremiumOf = (rd) => (rd.premiums ? +(rd.premiums[$('carbonGrade').value] || 0) : 0);
  const fmtMW = (v) => (v == null ? '—' : v >= 100 ? fmt(v, 0) : fmt(v, 1));
  const n3 = (v) => Number(v).toLocaleString('ko-KR', { maximumFractionDigits: 3 });

  function renderCaseRows() {
    const h = escapeHtml;
    const inp = (pfx, id, fld, v, step, label, ph) => `<input id="${pfx}-${h(id)}-${fld}" type="number" value="${v ?? ''}" step="${step}" min="0" placeholder="${h(ph)}" aria-label="${h(label)}">`;
    $('roundInputBody').innerHTML = ROUND_LIST.map((rd) => `
      <tr data-rd-row="${h(rd.id)}">
        <th class="mk-name">${h(rd.label)}<span class="sub">공고 ${fmtMW(rd.announcedMW)} / 접수 ${fmtMW(rd.receivedMW)} / 선정 ${fmtMW(rd.selectedMW)} MW</span></th>
        <td>${fmt(rd.cap, 3)}</td>
        <td>${inp('rd', rd.id, 'price', rd.avg ?? rd.cap, 0.01, `${rd.label} 적용 입찰가`, '상한가')}</td>
        <td>${inp('rd', rd.id, 'prem', '', 0.5, `${rd.label} 탄소 우대가격`, rd.premiums ? '자동' : '없음')}</td>
        <td>${inp('rd', rd.id, 'cf', '', 0.1, `${rd.label} 1년차 이용률`, '현재')}</td>
      </tr>`).join('');
    $('roundResultBody').innerHTML = ROUND_LIST.map((rd) => `
      <tr data-rd-row="${h(rd.id)}">
        <th class="mk-name">${h(rd.label)}<span class="sub">${rd.avg != null ? `낙찰 평균가 ${fmt(rd.avg, 3)}원` : '결과 발표 전 — 상한가 기준'}</span></th>
        <td id="rd-${h(rd.id)}-recv">—</td><td id="rd-${h(rd.id)}-pirr">—</td><td id="rd-${h(rd.id)}-eirr">—</td><td id="rd-${h(rd.id)}-gap">—</td><td id="rd-${h(rd.id)}-cappirr">—</td>
      </tr>`).join('');
    $('caseInputBody').innerHTML = CASE_LIST.map((c) => `
      <tr data-cs-row="${h(c.id)}">
        <th class="mk-name">${h(c.name)}<span class="sub">${fmtMW(c.mw)}MW · ${h(c.type)}</span></th>
        <td>${inp('cs', c.id, 'capex', c.capexPerMW, 0.1, `${c.name} 총사업비`, c.epcAmount ? 'EPC 역산' : '현재')}</td>
        <td>${inp('cs', c.id, 'cf', caseCfDefault(c), 0.1, `${c.name} 1년차 이용률`, '현재')}</td>
        <td>${inp('cs', c.id, 'r', '', 0.1, `${c.name} 수령단가 가정`, '상한가+우대')}</td>
      </tr>`).join('');
    $('caseResultBody').innerHTML = CASE_LIST.map((c) => `
      <tr data-cs-row="${h(c.id)}">
        <th class="mk-name">${h(c.name)}<span class="sub">${h(c.capexBasis)}${caseCfDefault(c) != null ? (c.cf != null ? ' · 이용률 보도' : ' · 발전량 환산') : ''}</span></th>
        <td id="cs-${h(c.id)}-inv">—</td><td id="cs-${h(c.id)}-need">—</td><td id="cs-${h(c.id)}-pirr">—</td><td id="cs-${h(c.id)}-eirr">—</td><td id="cs-${h(c.id)}-fit">—</td>
      </tr>`).join('');
    const part = (label, v) => (v ? `<br><span class="sub" style="display:inline;">${label}</span> ${h(v)}` : '');
    $('roundSourceList').innerHTML = ROUND_LIST.map((rd) => {
      const flow = [rd.receivedMW != null ? `접수 ${n3(rd.receivedMW)}MW${rd.receivedCount ? `(${n3(rd.receivedCount)}개소)` : ''} — 공고의 ${fmt(rd.receivedMW / rd.announcedMW * 100, 1)}%` : '',
        rd.selectedMW != null ? `선정 ${n3(rd.selectedMW)}MW${rd.selectedCount ? `(${n3(rd.selectedCount)}개소)` : ''}` : (rd.selectedCount ? `선정 ${n3(rd.selectedCount)}개소` : '')].filter(Boolean).join(' · ');
      return `<li data-rd-detail="${h(rd.id)}"><strong>${h(rd.label)}</strong> — 공고 ${n3(rd.announcedMW)}MW · 상한가 ${fmt(rd.cap, 3)}원/kWh${rd.avg != null ? ` · 낙찰 평균가 ${fmt(rd.avg, 3)}원/kWh` : ''}${rd.premiums ? ` · 탄소 우대 1등급 ${rd.premiums['1']}원 · 2등급 ${rd.premiums['2']}원` : ''}`
        + part('일정', rd.period) + part('접수 · 선정', flow) + part('참고', rd.note) + part('출처', rd.sources) + '</li>';
    }).join('');
    $('caseSourceList').innerHTML = CASE_LIST.map((c) => {
      const cfd = caseCfDefault(c);
      const gen = c.genGWh ? `연 ${n3(c.genGWh)}GWh → 1년차 이용률 약 ${fmt(cfd, 1)}%` : '';
      return `<li data-cs-detail="${h(c.id)}"><strong>${h(c.name)}</strong> — ${n3(c.mw)}MW · ${h(c.type)}`
        + part('사업자', c.developer) + part('모듈', c.module) + part('EPC·시공', c.epc) + part('사업비', c.capexText) + part('발전량', gen)
        + part('입지', c.siteText) + part('현황', c.status) + part('참고', c.note)
        + part('환경영향평가(EIASS)', c.eiass || 'EIASS 사업 검색에서 확인되지 않음(다른 사업명으로 등록됐을 가능성)') + part('출처', c.sources) + '</li>';
    }).join('');
  }

  function renderCases() {
    const base = readModel(), rv = base.revenue, n = rv.operatingRatesPct.length || 1;
    const d = rate(+$('degradationPct').value || 0);
    const cap0 = +$('priceCapPerKWh').value || 0, prem0 = carbonGradeTable()[$('carbonGrade').value]?.premium || 0;
    const ratesOf = (cf) => (cf > 0 ? Array.from({ length: n }, (_, i) => cf * (1 - d) ** i) : rv.operatingRatesPct.slice());
    // 사례는 고정가격계약(정액)으로 본다 — 페이지가 기업PPA 조합이어도 상승률 없이 계산
    const asFixed = (m) => { m.revenue.salesStructure = 'fixed'; m.revenue.contractEscalationPct = 0; return m; };
    const set = (id, text, cls = '') => { const el = $(id); el.textContent = text; el.className = cls; };
    const pct2 = (v) => (Number.isFinite(v) ? `${fmt(v * 100, 2)}%` : '—');
    const good = (r) => r && !(r.errors && r.errors.length);

    // ① 회차 — 우리 사업(현재 페이지)을 그 회차 가격으로 낙찰받았다고 보고 계산
    const roundPirrs = [];
    let roundMet = 0;
    ROUND_LIST.forEach((rd) => {
      const val = (fld) => { const v = $(`rd-${rd.id}-${fld}`).value; return v === '' ? NaN : +v; };
      const price = val('price') > 0 ? val('price') : (rd.avg ?? rd.cap), prem = val('prem') >= 0 ? val('prem') : roundPremiumOf(rd);
      const run = (R) => { const m = asFixed(cloneModel(base)); m.revenue.operatingRatesPct = ratesOf(val('cf')); m.revenue.bidPricePerKWh = R; return analyzeEss(m); };
      const r = run(price + prem);
      set(`rd-${rd.id}-recv`, `${fmt(price + prem, 2)}원`);
      if (!good(r)) { ['pirr', 'eirr', 'gap', 'cappirr'].forEach((fld) => set(`rd-${rd.id}-${fld}`, '산정 불가')); return; }
      set(`rd-${rd.id}-pirr`, pct2(r.projectIrr));
      set(`rd-${rd.id}-eirr`, pct2(r.equityIrr));
      const g = irrGapOf(r);
      if (g >= 0) roundMet++;
      set(`rd-${rd.id}-gap`, Number.isFinite(g) ? `${g >= 0 ? '+' : '−'}${fmt(Math.abs(g) * 100, 2)}%p` : '—', g >= 0 ? 'best' : 'over');
      const rc = run(rd.cap + prem);
      set(`rd-${rd.id}-cappirr`, good(rc) ? pct2(rc.projectIrr) : '산정 불가');
      roundPirrs.push(r.projectIrr);
    });

    // ② 대형 사업 — 사업비·이용률을 사례 값으로 바꿔 목표 IRR 적정 수령단가를 역산하고, 가정 단가의 수익률을 계산
    const curMW = rv.contractCapacityMW || 1, isEpc = (it) => it.group === 'construction';
    const baseInv = solveInvestment(base, buildCalendar(base.project)).totalInvestment;
    // 보도 총사업비(억원/MW)를 현재 페이지의 사업비 구성으로 공사비에 환산 — ① 공사비 ÷ 총투자비 비중
    const epcShare = baseInv > 0 ? sum(base.capex.items.filter(isEpc).map((it) => it.value)) / baseInv : 0.8;
    const needs = [];
    let fitCount = 0;
    CASE_LIST.forEach((c) => {
      const val = (fld) => { const v = $(`cs-${c.id}-${fld}`).value; return v === '' ? NaN : +v; };
      // 총사업비 입력이 없으면: EPC 금액이 있으면 그 금액을 ① 공사비로 보고 총사업비로 역산, 없으면 현재 페이지 가정
      const epcTotal = val('capex') > 0 ? val('capex') * c.mw * epcShare : c.epcAmount ? c.epcAmount : (baseInv / curMW) * c.mw * epcShare;
      const ratio = c.mw / curMW, R = val('r') > 0 ? val('r') : cap0 + prem0;
      const build = (price) => {
        const m = asFixed(cloneModel(base));
        m.revenue.contractCapacityMW = c.mw;
        m.revenue.operatingRatesPct = ratesOf(val('cf'));
        m.capex = { ...m.capex, items: [{ id: 'epcLumpSum', label: `EPC (${c.name})`, value: epcTotal, category: '기자재', group: 'construction' },
          ...m.capex.items.filter((it) => !isEpc(it)).map((it) => ({ ...it, value: it.value * ratio }))] };
        m.opex = { ...m.opex, items: m.opex.items.map((it) => ({ ...it, value: it.value * ratio })) };
        m.finance = { ...m.finance, fundingMode: 'ratio' };
        m.revenue.bidPricePerKWh = price;
        return m;
      };
      const r = analyzeEss(build(R));
      if (!good(r)) { ['inv', 'need', 'pirr', 'eirr', 'fit'].forEach((fld) => set(`cs-${c.id}-${fld}`, '산정 불가')); return; }
      set(`cs-${c.id}-inv`, fmt(r.totalInvestment / c.mw, 1));
      set(`cs-${c.id}-pirr`, pct2(r.projectIrr));
      set(`cs-${c.id}-eirr`, pct2(r.equityIrr));
      const sv = solveBidPriceForTarget(build(0), rate(+$('targetIrrPct').value || 0), $('irrKind').value);
      if (!sv.ok) { set(`cs-${c.id}-need`, '500원으로도 미달', 'over'); set(`cs-${c.id}-fit`, '—'); return; }
      needs.push(sv.price);
      set(`cs-${c.id}-need`, `${fmt(sv.price, 2)}원`);
      const bid = sv.price - prem0, fits = cap0 > 0 && bid <= cap0;
      if (fits) fitCount++;
      set(`cs-${c.id}-fit`, cap0 > 0 ? (fits ? `가능 (${fmt(cap0 - bid, 2)}원 여유)` : `${fmt(bid - cap0, 2)}원 초과`) : '—', fits ? 'best' : 'over');
    });

    const spread = (arr, f) => {
      const s = [...arr].sort((a, b) => a - b);
      return s.length ? { lo: f(s[0]), hi: f(s[s.length - 1]), med: f(s[Math.floor((s.length - 1) / 2)]) } : null;
    };
    const rp = spread(roundPirrs, pct2), np = spread(needs, (v) => `${fmt(v, 2)}원`);
    $('caseSummary').innerHTML = [
      rp ? `<strong>① 회차 가격으로 우리 사업을 낙찰받았다면 P-IRR ${rp.lo} ~ ${rp.hi}</strong> — ${irrTargetLabel()} 달성 ${roundMet}/${ROUND_LIST.length}회차` : '',
      np ? `<strong>② 대형 사업 ${needs.length}건의 ${irrTargetLabel()} 적정 수령단가 ${np.lo} ~ ${np.hi} (중앙값 ${np.med})</strong> — 현재 상한가 ${fmt(cap0, 3)}원 + 우대 ${fmt(prem0, 0)}원 기준 입찰 가능 ${fitCount}건` : '',
    ].filter(Boolean).join('<br>') || '사례를 계산하지 못했습니다.';
  }
